    接入阿里云serverless rds

    1. 创建数据库实例

    1.1 购买数据库实例并配置

    首先需要去阿里云工作台购买rds数据库，这里购买的是计费方式为serverless的(不访问的时候自动停止不收费，针对博客类比较合适)，
    底层数据库使用MySQL8.0(开源，更通用，资料也多)

    1.2 在数据库实例创建业务所需要的表

    CREATE DATABASE album
DEFAULT CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

    建表

    use album;

CREATE TABLE
  photos (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY COMMENT '图片唯一ID',
    user_id BIGINT UNSIGNED DEFAULT NULL COMMENT '用户ID',
    oss_key VARCHAR(512) NOT NULL COMMENT 'OSS对象路径',

    -- 文件信息
    file_name VARCHAR(512) DEFAULT NULL COMMENT '文件名',
    file_type VARCHAR(32) DEFAULT NULL COMMENT 'jpg/heic/png/mp4',
    file_size BIGINT UNSIGNED DEFAULT NULL COMMENT '文件大小(byte)',
    -- 时间信息
    upload_time DATETIME NOT NULL COMMENT '上传OSS时间',
    capture_time DATETIME DEFAULT NULL COMMENT 'EXIF拍摄时间',
    sort_time DATETIME NOT NULL COMMENT '最终排序时间',
    -- 图片信息
    width INT DEFAULT NULL COMMENT '宽',
    height INT DEFAULT NULL COMMENT '高',
    orientation INT DEFAULT NULL COMMENT '旋转方向',
    -- EXIF信息
    camera_make VARCHAR(128) DEFAULT NULL COMMENT '相机厂商',
    camera_model VARCHAR(128) DEFAULT NULL COMMENT '相机型号',
    -- GPS
    latitude DECIMAL(10, 7) DEFAULT NULL COMMENT '纬度',
    longitude DECIMAL(10, 7) DEFAULT NULL COMMENT '经度',
    -- 扩展字段
    metadata JSON DEFAULT NULL COMMENT '扩展metadata',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_oss_key (oss_key),
    INDEX idx_sort_time (sort_time DESC),
    INDEX idx_capture_time (capture_time),
    INDEX idx_upload_time (upload_time)
  ) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci ROW_FORMAT = DYNAMIC COMMENT = '相册图片索引表';

    2. 函数fc和角色创建

    2.1 部署函数fc

    在env.example配置好阿里云相关的账户密钥等信息之后执行命令行执行以下命令创建阿里云函数fc
    创建完成之后可以直接到阿里云工作台查看

    cd functions/oss-photo-sync
npm install -g @serverless-devs/s
source env.example
# 首次配置阿里云凭证，别名选择 default
s config add
s build --use-docker
s deploy

    2.2 创建并配置指定的角色和权限策略

    这里函数fc涉及到的oss触发和rds数据库操作需要不同的角色和权限 需要去阿里云工作台进行创建，参考如下 notification信任策略

    {
  "Statement": [
    {
      "Action": "sts:AssumeRole",
      "Effect": "Allow",
      "Principal": {
        "Service": [
          "oss.aliyuncs.com"
        ]
      }
    }
  ],
  "Version": "1"
}

    sync execution role的信任策略

    {
  "Statement": [
    {
      "Action": "sts:AssumeRole",
      "Effect": "Allow",
      "Principal": {
        "Service": [
          "fc.aliyuncs.com"
        ]
      }
    }
  ],
  "Version": "1"
}

    两个自定义权限策略内容 OssPhotoSyncPassRolePolicy

    {
  "Version": "1",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": "ram:PassRole",
      "Resource": [
        "acs:ram::1487190xxxxx:role/oss-photo-sync-execution-role",
        "acs:ram::1487190xxxxx:role/aliyunosseventnotificationrole"
      ]
    },
    {
      "Effect": "Allow",
      "Action": [
        "vpc:DescribeVSwitchAttributes",
        "vpc:DescribeVpcAttribute",
        "vpc:DescribeVSwitches",
        "vpc:DescribeVpcs",
        "ecs:DescribeSecurityGroups"
      ],
      "Resource": "*"
    }
  ]
}

    OssPhotoSyncRuntimePolicy

    {
  "Version": "1",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": "oss:ListObjects",
      "Resource": "acs:oss:*:*:1024daniel"
    },
    {
      "Effect": "Allow",
      "Action": "oss:GetObject",
      "Resource": [
        "acs:oss:*:*:1024daniel/album/*",
        "acs:oss:*:*:1024daniel/dishes/*"
      ]
    },
    {
      "Effect": "Allow",
      "Action": "oss:PutObject",
      "Resource": "acs:oss:*:*:1024daniel/manifests/*"
    }
  ]
}
